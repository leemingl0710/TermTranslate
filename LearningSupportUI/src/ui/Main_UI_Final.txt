private JPanel createOCRPage() {

    JPanel panel = new JPanel(new BorderLayout(15, 15));
    panel.setBackground(new Color(245, 248, 253));
    panel.setBorder(
        BorderFactory.createEmptyBorder(25, 30, 25, 30)
    );

    JLabel heading = new JLabel("OCR Text Review");
    heading.setFont(new Font("Segoe UI", Font.BOLD, 26));
    heading.setForeground(new Color(35, 55, 90));

    JLabel subtitle = new JLabel(
        "Review and correct the extracted text before analysis."
    );
    subtitle.setFont(new Font("Segoe UI", Font.PLAIN, 14));
    subtitle.setForeground(new Color(100, 110, 125));

    JPanel topPanel = new JPanel(new GridLayout(2, 1, 5, 5));
    topPanel.setBackground(new Color(245, 248, 253));

    topPanel.add(heading);
    topPanel.add(subtitle);

    panel.add(topPanel, BorderLayout.NORTH);

    ocrTextArea = new JTextArea();
    ocrTextArea.setFont(new Font("Malgun Gothic", Font.PLAIN, 15));
    ocrTextArea.setLineWrap(true);
    ocrTextArea.setWrapStyleWord(true);
    ocrTextArea.setEditable(true);
    ocrTextArea.setBorder(
        BorderFactory.createEmptyBorder(15, 15, 15, 15)
    );

    JScrollPane scrollPane = new JScrollPane(ocrTextArea);
    panel.add(scrollPane, BorderLayout.CENTER);

    JButton confirmButton = new JButton("Confirm Text");
    JButton backButton = new JButton("Back to Upload");

    JButton[] buttons = {backButton, confirmButton};

    for (JButton button : buttons) {
        button.setFont(new Font("Segoe UI", Font.BOLD, 14));
        button.setBackground(new Color(45, 90, 150));
        button.setForeground(Color.WHITE);
        button.setFocusPainted(false);
        button.setOpaque(true);
        button.setBorderPainted(false);
        button.setPreferredSize(new Dimension(180, 42));
    }

    JPanel buttonPanel = new JPanel(
        new FlowLayout(FlowLayout.CENTER, 15, 10)
    );
    buttonPanel.setBackground(new Color(245, 248, 253));

    buttonPanel.add(backButton);
    buttonPanel.add(confirmButton);

    panel.add(buttonPanel, BorderLayout.SOUTH);

    backButton.addActionListener(e -> showPage("UPLOAD"));

    confirmButton.addActionListener(e -> {

        if (ocrTextArea.getText().trim().isEmpty()) {
            JOptionPane.showMessageDialog(
                panel, "There is no text to confirm."
            );
            return;
        }

        JOptionPane.showMessageDialog(
            panel,
            "Text confirmed. Ready for analysis."
        );
    });

    return panel;
}


private JPanel createHistoryPage() {
	JPanel panel = new JPanel(new BorderLayout(20,20));

	panel.setBorder(BorderFactory.createEmptyBorder(25,30,25,30));
	
	JLabel heading = new JLabel("Analysis History");

	heading.setForeground(new Color(35,55,90));
	
	
	JLabel subtitle = new JLabel("View your previously analyzed learning materials.");
	subtitle.setFont(new Font("Segoe UI",Font.PLAIN,14));
	subtitle.setForeground(new Color(100,110,125));
	
	JPanel topPanel = new JPanel(new GridLayout(2,1,5,5));
	topPanel.setBackground(new Color(245,248,253));
	topPanel.add(heading);
	topPanel.add(subtitle);

	JTextField searchField = new JTextField(20);
	searchField.setFont(new Font("Malgun Gothic", Font.PLAIN, 14));

	JButton searchButton = new JButton("Search");
	JButton resetButton = new JButton("Reset");

	JPanel searchPanel = new JPanel(
	    new FlowLayout(FlowLayout.LEFT, 10, 5)
	);

	searchPanel.setBackground(new Color(245, 248, 253));

	searchPanel.add(new JLabel("Search File:"));
	searchPanel.add(searchField);
	searchPanel.add(searchButton);
	searchPanel.add(resetButton);

	JPanel headerPanel = new JPanel(new BorderLayout(5, 10));
	headerPanel.setBackground(new Color(245, 248, 253));

	headerPanel.add(topPanel, BorderLayout.NORTH);
	headerPanel.add(searchPanel, BorderLayout.SOUTH);

	panel.add(headerPanel, BorderLayout.NORTH);
	
	JPanel historyPanel = new JPanel();
	historyPanel.setLayout(new BoxLayout(historyPanel, BoxLayout.Y_AXIS));
	historyPanel.setBackground(new Color(245,248,253));
	
	String[] files = {
			"Introduction to Data Science.pdf",
			"Database Lecture 03.pptx",
			"Computer Advanced Math.pdf"
	};
	
	for(String file:files) {
		JPanel item = new JPanel(new BorderLayout(10,10));
		item.setMaximumSize(new Dimension(Integer.MAX_VALUE, 85));
		item.setBackground(Color.WHITE);
		item.setBorder(BorderFactory.createCompoundBorder(
				BorderFactory.createLineBorder(new Color(225,232,242),1
						),
				BorderFactory.createEmptyBorder(15,20,15,20)
				));
		
		JLabel fileName = new JLabel(file);
		fileName.setFont(new Font("Segoe UI",Font.BOLD,15));
		fileName.setForeground(new Color(35,33,90));
		
		JLabel status = new JLabel("Analysis completed");
		status.setFont(new Font("Segoe UI",Font.PLAIN,12));
		status.setForeground(new Color(70,145,110));
		
		JButton viewButton = new JButton("View");
		viewButton.setFont(new Font("Segoe UI", Font.BOLD, 13));
		viewButton.setBackground(new Color(45, 90, 150));
		viewButton.setForeground(Color.WHITE);
		viewButton.setFocusPainted(false);
		viewButton.setOpaque(true);
		viewButton.setBorderPainted(false);
		viewButton.setPreferredSize(new Dimension(90, 35));
		
		JPanel infoPanel = new JPanel(new GridLayout(2,1,5,5));
		infoPanel.setBackground(Color.WHITE);
		infoPanel.add(fileName);
		infoPanel.add(status);
		
		JButton deleteButton = new JButton("Delete");

		deleteButton.setFont(new Font("Segoe UI", Font.BOLD, 13));
		deleteButton.setBackground(new Color(180, 65, 65));
		deleteButton.setForeground(Color.WHITE);
		deleteButton.setFocusPainted(false);
		deleteButton.setOpaque(true);
		deleteButton.setBorderPainted(false);
		deleteButton.setPreferredSize(new Dimension(90, 35));

		JPanel actionPanel = new JPanel(
		    new FlowLayout(FlowLayout.RIGHT, 8, 0)
		);

		actionPanel.setBackground(Color.WHITE);
		actionPanel.add(viewButton);
		actionPanel.add(deleteButton);

		item.add(infoPanel, BorderLayout.CENTER);
		item.add(actionPanel, BorderLayout.EAST);

		deleteButton.addActionListener(e -> {

		    int choice = JOptionPane.showConfirmDialog(
		        this,
		        "Do you want to delete this history record?\n" + file,
		        "Delete History",
		        JOptionPane.YES_NO_OPTION
		    );

		    if (choice == JOptionPane.YES_OPTION) {

		        historyPanel.remove(item);
		        historyPanel.revalidate();
		        historyPanel.repaint();
		    }
		});
		
		viewButton.addActionListener(e -> {
			showPage("RESULT");
		}
				);
		
		historyPanel.add(item);
		historyPanel.add(Box.createVerticalStrut(12));
		
		}
	
	searchButton.addActionListener(e -> {

	    String keyword = searchField.getText()
	        .trim().toLowerCase();

	    for (Component component : historyPanel.getComponents()) {

	        if (component instanceof JPanel) {

	            JPanel item = (JPanel) component;

	            JPanel infoPanel = (JPanel)
	                item.getComponent(0);

	            JLabel fileLabel = (JLabel)
	                infoPanel.getComponent(0);

	            item.setVisible(
	                fileLabel.getText().toLowerCase()
	                    .contains(keyword)
	            );
	        }
	    }

	    historyPanel.revalidate();
	    historyPanel.repaint();
	});

	resetButton.addActionListener(e -> {
	    searchField.setText("");
	    searchButton.doClick();
	});

	searchField.addActionListener(e -> searchButton.doClick());
	
	JScrollPane scrollPane = new JScrollPane(historyPanel);
	scrollPane.setBorder(null);
	scrollPane.getViewport().setBackground(new Color(245,248,253));
	panel.add(scrollPane,BorderLayout.CENTER);
	
	return panel;
	
}

private void showPage(String pageName) {
    cardLayout.show(contentPanel, pageName);
}

private void createChapter() {

    String chapterName = JOptionPane.showInputDialog(
        this,
        "Enter a chapter name:",
        "Create Chapter",
        JOptionPane.PLAIN_MESSAGE
    );

    if (chapterName == null) {
        return;
    }

    chapterName = chapterName.trim();

    if (chapterName.isEmpty()) {
        JOptionPane.showMessageDialog(
            this, "Please enter a chapter name."
        );
        return;
    }

    if (chapters.contains(chapterName)) {
        JOptionPane.showMessageDialog(
            this, "This chapter already exists."
        );
        return;
    }

    chapters.add(chapterName);
    chapterBox.addItem(chapterName);
    chapterBox.setSelectedItem(chapterName);

    JOptionPane.showMessageDialog(
        this,
        "Chapter created: " + chapterName
    );
}


public List<File> getSelectedFiles() {
    return new ArrayList<>(selectedFiles);

public String getSelectedChapter() {
    if (chapterBox.getSelectedIndex() <= 0) {
        return null;
    }

    return (String) chapterBox.getSelectedItem();
}

public void updateAnalysisProgress(int progress) {
    SwingUtilities.invokeLater(() -> {
        analysisProgressBar.setValue(progress);
        analysisProgressBar.setString(
            "Analysis Progress: " + 
            analysisProgressBar.getValue() + "%"
        );
    });
}

public void showAnalysisError(String errorMessage, Runnable retryAction) {

    Object[] options = {"Retry", "Cancel"};

    int choice = JOptionPane.showOptionDialog(
        this,
        errorMessage + "\nWould you like to try again?",
        "Analysis Failed",
        JOptionPane.YES_NO_OPTION,
        JOptionPane.ERROR_MESSAGE,
        null,
        options,
        options[0]
    );

    if (choice == 0 && retryAction != null) {
        retryAction.run();
    }
}
